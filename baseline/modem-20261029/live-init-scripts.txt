#!/bin/sh /etc/rc.common
# Android vendor services in a chroot: modem_control (PM watchdog + modem boot), cp_diskserver (modem NV), refnotify
START=11
USE_PROCD=1
export PATH="$PATH:/opt/changxing60/busybox-bin"

start_service() {
	# nothing to boot on a kernel without the vendor modem loader (mainline)
	[ -e /dev/modem ] || [ -e /dev/pmsys ] || { logger -t changxing60-vendor "no modem stack in $(uname -r)"; return 0; }
	procd_open_instance modem_control
	procd_set_param command /opt/changxing60/bin/android-vendor-start
	procd_set_param env PATH="$PATH"
	procd_set_param respawn 3600 10 0
	procd_set_param stdout 1
	procd_set_param stderr 1
	procd_close_instance
	for d in cp_diskserver refnotify; do
		procd_open_instance "$d"
		procd_set_param command /bin/sh -c "sleep 12; exec /opt/changxing60/bin/android-run $d"
		procd_set_param env PATH="$PATH"
		procd_set_param respawn 3600 10 0
		procd_close_instance
	done
}
#!/bin/sh /etc/rc.common
# Keep the modem's AT channels open for the lifetime of the boot.
#
# /dev/stty_nr* are SIPC channels, not serial ports: only one reader gets the data, and closing one while the
# modem runs leaves it silent for ever while still occupying the device -- only a reboot brings it back (see
# changxing60-atd for the measurements). Android's RIL holds nr0-nr5 open for the whole life of the system. Until
# this service existed nothing did, so every AT user opened the channel for itself, and whenever two of them
# were alive at once each got a random half of every reply. That is the shape of "the modem stopped answering
# after a few minutes": not a dead modem, two readers.
#
# nr1 is the shared command channel. mobile-data's dial, the SMS stack (changxing60-smsd), the status page and the
# voice control all reach it through changxing60-at and the daemon's command fifo, one command at a time.
#
# nr2 is mobile-data's own channel, and it sits behind a switch (/etc/changxing60/at-split) rather than being always
# on. Why it exists: AT+CGDATA blocks its channel for 10-20 s while the bearer comes up, and on a shared nr1
# every signal check and SMS waited behind that. Why it is not the default: issuing the data call on nr2 is an
# unmeasured change on this modem -- our dial has only ever run on nr1 -- and the wrong channel means an attach
# that never returns, i.e. no WAN at all. So the default is one daemon on nr1, which is already enough to give
# every AT user a single owner, and the split is something to turn on and measure.
#
# START=19, not 91. netifd (S20network) runs the wan protocol, and that dial is nothing but AT commands, all
# of them through these daemons. Started later, the first attach found no daemon, opened nr1 directly ("works
# once, and then the channel needs a modem restart"), and then the daemon opened the same channel again: two
# readers on one SIPC tty.
START=19
USE_PROCD=1
export PATH="$PATH:/opt/changxing60/busybox-bin"

# Wait for the SIPC node inside the procd command, never in start_service: the nodes appear only once the CP
# is alive, and an init script that returns without opening an instance never gets a second chance. Bounded
# (240 s) so that a kernel without the modem stack says so instead of spinning silently; procd respawns and
# retries, which also covers the CP coming up late.
wait_and_exec() {  # wait_and_exec TTYNODE DAEMON...; the daemon's environment comes from procd_set_param env
	local dev=$1; shift
	echo "n=0; while [ ! -c $dev ]; do n=\$((n+1)); [ \"\$n\" -gt 240 ] && { echo 'changxing60-at: $dev never appeared' >&2; exit 1; }; sleep 1; done; exec $*"
}

start_service() {
	procd_open_instance atd
	procd_set_param command /bin/sh -c "$(wait_and_exec /dev/stty_nr1 /opt/changxing60/bin/changxing60-atd)"
	procd_set_param respawn 3600 10 0
	procd_set_param stdout 1
	procd_set_param stderr 1
	procd_close_instance

	# mobile-data's own channel. Created by hand on the device (echo > /etc/changxing60/at-split) to be measured:
	# without it mobile-data routes its dial through nr1 like every other client.
	[ -e /etc/changxing60/at-split ] || return 0
	# No URC channels on this one: nr0's unsolicited stream belongs to the daemon above, and a second reader
	# would split it between them. CHANGXING60_AT_URC_CHANNELS is set empty on purpose -- changxing60-atd reads a value
	# that is set but empty as "drain nothing", where an unset one would fall back to nr0.
	procd_open_instance atd2
	procd_set_param command /bin/sh -c "$(wait_and_exec /dev/stty_nr2 /opt/changxing60/bin/changxing60-atd)"
	procd_set_param env CHANGXING60_AT_DEV=/dev/stty_nr2 CHANGXING60_AT_DIR=/run/changxing60-at2 CHANGXING60_AT_URC_CHANNELS=
	procd_set_param respawn 3600 10 0
	procd_set_param stdout 1
	procd_set_param stderr 1
	procd_close_instance
}
#!/bin/sh /etc/rc.common
# After boot: keep mobile data up across modem resets, and mark the boot successful (re-arm slot b if Linux is default)
START=99
USE_PROCD=1
export PATH="$PATH:/opt/changxing60/busybox-bin"

start_service() {
	case "$(uname -r)" in
		6.*)
			logger -t changxing60-post "6.x: vendor modem/WiFi watchers skipped (no vendor modem/WCN stack)"
			return 0
			;;
	esac
	# Why did this boot happen? Write it down. /proc/cmdline's bootcause is generated per boot by the
	# bootloader, and pstore keeps only ONE crash record, so without this line a reboot's cause is gone
	# the moment the next one lands. That is not hypothetical: on 2026-10-16 an oops in the audio path
	# (see docs/changxing60/audio-oops.md) took the device down at 18:28, and an earlier reboot at 19:49
	# in the same evening could not be explained at all -- its pstore record had already been overwritten.
	# One appended line per boot makes both answerable. Append-only, capped, and it must never be the
	# thing that fails a boot, hence the redirect and the trailing true.
	{
		hist=/etc/changxing60/boot-history
		mkdir -p /etc/changxing60 2>/dev/null
		printf '%s uptime=%s kernel=%s %s\n' \
			"$(date '+%Y-%m-%d %H:%M:%S %z')" \
			"$(cut -d. -f1 /proc/uptime 2>/dev/null)" \
			"$(uname -r 2>/dev/null)" \
			"$(grep -o 'bootcause="[^"]*"' /proc/cmdline 2>/dev/null) $(grep -o 'androidboot\.mode=[^ ]*' /proc/cmdline 2>/dev/null)" \
			>> "$hist" 2>/dev/null
		tail -n 200 "$hist" > "$hist.new" 2>/dev/null && mv "$hist.new" "$hist" 2>/dev/null
	} 2>/dev/null
	# Last word on the console level: changxing60-hw set it before the WLAN modules loaded, but the
	# vendor modules keep loading after that (S12 is early), so re-assert it here. See quiet-console
	# for why a chatty console wedges the SC2355 firmware.
	/opt/changxing60/bin/quiet-console
	procd_open_instance cellwatch
	procd_set_param command /opt/changxing60/bin/mobile-data watch
	procd_set_param env CHANGXING60_NETIFD=1 PATH="$PATH"
	procd_set_param respawn 3600 30 0
	procd_close_instance
	procd_open_instance thermalguard
	procd_set_param command /opt/changxing60/bin/thermal-guard
	procd_set_param env PATH="$PATH"
	procd_close_instance
	# The bq2560x probe programs the end-of-charge threshold from the pack's *max charge
	# current* (charge stops at ~94%) and sizes the input current limit from USB detection
	# (0.19-0.8 A into the cell depending on the boot), so the cell never charged at its
	# ~1 A. charge-fix re-pins ITERM 180 mA / input limit 1.2 A / ICHG 960 mA / VREG
	# 4368 mV (the vendor driver's own value for this pack; the live chip was once found
	# at 4432 mV from a one-shot write of unknown origin) on the chip over i2c, and
	# applies the optional /opt/changxing60/charge-cap top-off cap (charge off at the cap,
	# back on 40 permille below; the board keeps running off VBUS either way). A
	# crontab entry re-runs it every minute against USB replugs. Waiting for
	# the i2c node is the script's job.
	procd_open_instance chargefix
	procd_set_param command /opt/changxing60/bin/charge-fix
	procd_set_param env PATH="$PATH"
	procd_close_instance
	# The SC2355 sometimes refuses the first IFF_UP (EIO), and an AP that came up while the world
	# regulatory domain was still active stays AP-DISABLED: both need a real down/up. This has to run
	# after netifd (S20) -- changxing60-hw is S12 and its wifi calls there only answer "Command failed: Not
	# found" -- so it lives here, 35 s in, with wififix deciding whether anything is wrong.
	procd_open_instance wififix
	procd_set_param command /bin/sh -c "sleep 35; exec /opt/changxing60/bin/wifi-fix"
	procd_set_param env PATH="$PATH"
	procd_close_instance
	# wifi-fix repairs the AP once, at startup. This keeps repairing it: an AP that goes off air
	# later (a firmware assert that powered the radio off under changxing60-hw's reset policy, a failed
	# CMD_START_AP) looks perfectly healthy from the host, so something has to test it the way a
	# client would and put it back. Restarts wifi rather than rebooting -- see hotspot-watch.
	procd_open_instance hotspotwatch
	procd_set_param command /opt/changxing60/bin/hotspot-watch
	procd_set_param env PATH="$PATH"
	procd_set_param respawn 3600 30 0
	procd_close_instance
	procd_open_instance bootok
	procd_set_param command /bin/sh -c "sleep 30; exec /opt/changxing60/bin/changxing60-next-boot --rearm"
	procd_set_param env PATH="$PATH"
	procd_close_instance
}
