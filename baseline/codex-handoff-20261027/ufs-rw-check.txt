#!/bin/sh /etc/rc.common
# Answer modem_control's CP resets, so an assert costs seconds instead of five minutes.
#
# modem_control (init.d/changxing60-vendor, S11) is the process that resets the CP after it asserts. It
# waits for a "SLOGMODEM DUMP COMPLETE" from the CP log daemon before it does, and this board has no such
# daemon, so every assert ran out its full 300 s timeout with the WAN down for all of it -- which is most
# of what "the WAN died again for a few minutes" has been. This is that daemon's client half: it answers at
# once, and keeps a log of what the modem was doing, because nothing else records it (the console is pinned
# to ERR and above and a modem state message is not a kernel error).
#
# START=12: one after changxing60-vendor (S11), which is what creates the socket. That is not a hard
# requirement -- the client retries until the socket appears, precisely so that a restart of modem_control
# does not need this restarted in step -- but starting it late means the first assert after boot is a slow
# one for no reason.
#
# procd supervises it: the client exits non-zero when the socket closes, which is what happens when
# modem_control is restarted, and respawn is how it reconnects. The binary's own reconnect loop covers the
# other case, modem_control not being up yet.
#
# The protocol and how it was verified against this board's own modem_control are in
# tools/modem/changxing60-modemd.c; the whole story is in docs/changxing60/telephony.md.
START=12
USE_PROCD=1
export PATH="$PATH:/opt/changxing60/busybox-bin"

start_service() {
        # same guard as changxing60-vendor: there is no modem stack to talk to on the mainline kernel
        [ -e /dev/modem ] || [ -e /dev/pmsys ] || { logger -t changxing60-modemd "no modem stack in $(uname -r)"; return 0; }
        [ -x /opt/changxing60/bin/changxing60-modemd ] || { logger -t changxing60-modemd "no binary"; return 0; }
        procd_open_instance modemd
        procd_set_param command /opt/changxing60/bin/changxing60-modemd
        procd_set_param respawn 3600 10 0
        procd_set_param stdout 1
        procd_set_param stderr 1
        procd_close_instance
}
