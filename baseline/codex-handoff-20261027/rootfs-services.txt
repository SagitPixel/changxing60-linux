/ # > #!/bin/sh /etc/rc.common
# Load the Unisoc ASoC / AGDSP audio driver set, so that card 0 (sprdphone-sc2730) exists at all.
#
# START=20: after init.d/changxing60-hw (S12), which fixes the Wi-Fi MAC and the other per-boot
# hardware bits, and before init.d/changxing60-voiced (S21), which is what will drive call audio
# once its voice-audio half exists. Nothing else in the boot depends on the sound card, so this is
# not on the critical path: a module that will not load is logged and the boot carries on.
#
# Not procd, because there is no process to supervise -- the loader loads what it can and exits.
# It is written to be safe to run again at any time (a module already in is left alone), which
# matters here more than usual: this board reboots itself when the WAN wedges, and a loader caught
# half way through must not be the thing that leaves the sound card missing for the next boot.
#
# The set is 27 modules and the whole story is in the loader and in docs/changxing60/telephony.md.
# If it fails, /root/audio-load.log says where it stopped; the console will not, because the
# console is pinned to ERR and above and a module that will not load is not a kernel error.
START=20
STOP=20

start() {
        # Order matters here, and it is the whole point of changxing60-audio-up: the 27 ASoC modules go in
        # first (the machine card defers), then the VBC profiles are written while that card is *not* bound
        # (writing them after it binds NULL-derefs in vbc_turning_ndp_write and panics), and only then is
        # sprd-dma loaded -- which is what lets sound@0 probe and the FE PCMs appear. See that script.
        #
        # CHANGXING60_VBC_LOAD=0 in /etc/config/changxing60-audio (a uci-free shell env, read here) skips
        # the profile step: card without DSP parameters, safe mode.
        if [ -x /opt/changxing60/bin/changxing60-audio-up ]; then
                /opt/changxing60/bin/changxing60-audio-up
        else
                logger -t changxing60-audio "no bring-up script at /opt/changxing60/bin/changxing60-audio-up"
        fi
        # ALSA's device nodes are made by the kernel, but /dev/snd is 0660 root:audio and nothing in
        # this rootfs is in the audio group -- the panel and the voice daemon run as root, so this is
        # only about the nodes existing at all when the kernel did not make them.
        [ -d /dev/snd ] || { mkdir -p /dev/snd 2>/dev/null; }
        return 0
}

stop() {
        # Unloading the set is not supported: snd-soc-sprd-card cannot be unbound from sound@0
        # without taking the whole ASoC core with it, and on this board a partially unloaded audio
        # stack is worse than a loaded one. Nothing calls this.
        return 0
}
#!/bin/sh /etc/rc.common
# Hardware bring-up before netifd: vendor drivers, zram, Wi-Fi (SC2355), first-boot Wi-Fi config, Bluetooth off
START=12
export PATH="$PATH:/opt/changxing60/busybox-bin"

boot() {
        /opt/changxing60/bin/extra-modules >/tmp/changxing60-extra-modules.log 2>&1
        # Vendor drivers Android loads that the boot set does not carry: CPU crypto offload, USB-Ethernet,
        # sensor hub, ION/DMA/DVFS glue. Staged from the same 5.4 build as the boot set, so the vermagic
        # matches; see docs/changxing60/drivers.md. The touchscreen is NOT one of them -- it stays opt-in
        # through touch-ctl, see the note in that script.
        /opt/changxing60/bin/vendor-modules >>/tmp/changxing60-vendor-modules.log 2>&1
        # charge-fix re-pins the bq2560x registers the vendor probe leaves at POR defaults (ITERM/VREG/
        # ICHG/IINLIM -- see the charging doc; the chip sat at 960 mA input on USB detection otherwise).
        # Like bootdiag below, this line used to live only in the hand-edited crontab of an already
        # installed device, so a fresh flash charged at ~0.5 A again and nothing reported it.
        if [ -f /etc/crontabs/root ] || [ -d /etc/crontabs ]; then
                grep -q 'changxing60/bin/charge-fix' /etc/crontabs/root 2>/dev/null || \
                        echo '* * * * * /opt/changxing60/bin/charge-fix >/dev/null 2>&1' >> /etc/crontabs/root
        fi
        # bootdiag writes the evidence a post-switch_root failure leaves nowhere else (WCN/hostapd/netifd,
        # the wedge watch) into /root/diag, and attaches a UDC that only turns up after the initramfs gave
        # up. It is driven by cron because a wedged init queue would stall an init script. Re-asserted
        # every boot: this rootfs' crontab is hand-edited, so an install from before it exists would
        # otherwise never get the line.
        if [ -f /etc/crontabs/root ] || [ -d /etc/crontabs ]; then
                grep -q 'changxing60/bin/bootdiag' /etc/crontabs/root 2>/dev/null || \
                        echo '* * * * * /opt/changxing60/bin/bootdiag >/dev/null 2>&1' >> /etc/crontabs/root
        fi
        # After the vendor modules are in (one of them is what leaves console_loglevel at 7 rather
        # than the quiet default the kernel was built and booted with) and before the WLAN ones, so
        # the SC2355 driver's startup burst never reaches the 115200-baud console. See quiet-console.
        /opt/changxing60/bin/quiet-console
        /opt/changxing60/bin/zram-start >/dev/null 2>&1
        /opt/changxing60/bin/wifi-start >/tmp/changxing60-wifi-start.log 2>&1
        # sprd_wlan_combo is loaded by wifi-start; re-assert in case its probe raised the level again.
        /opt/changxing60/bin/quiet-console
        # What the WCN framework should do when the firmware asserts. The build default is "dump"
        # (read out the firmware's memory and stop: is_reset = WCN_ASSERT_ONLY_DUMP, sysfs.h:15, set
        # from the FLAG_WCN_USER-less branch at sysfs.c:1064), which leaves the chip wedged with
        # cp_asserted = 1 until a reboot -- the radio stays up as far as hostapd is concerned and no
        # client gets a packet. "reset" instead powers the radio off on an assert and lets the WLAN
        # driver's own reset_work (iface.c:150 -> sc2355_reset_self, tx.c:1820) clear the wedge;
        # hotspot-watch then re-opens the interface, which re-downloads the firmware. This is a
        # kernel-memory atomic, so it is re-applied every boot, after the wcn misc device exists.
        R=/sys/devices/virtual/misc/wcn/devices/reset_dump
        i=0
        while [ $i -lt 30 ]; do
                [ -w "$R" ] && break
                sleep 1
                i=$((i + 1))
        done
        [ -w "$R" ] && echo reset > "$R"
        # cfg80211 looked for regulatory.db before the rootfs was mounted; load it now so hostapd sees the country rules
        /opt/changxing60/bin/regdb-load >>/tmp/changxing60-wifi-start.log 2>&1
        /opt/changxing60/bin/openwrt-wifi-config >>/tmp/changxing60-wifi-start.log 2>&1
        # apply the country before netifd starts hostapd, otherwise 5 GHz channels are still NO-IR under world rules
        country=$(uci -q get wireless.radio0.country)
        if [ -n "$country" ]; then
                iw reg set "$country"
                for n in $(seq 1 10); do iw reg get | grep -q "country $country:" && break; sleep 1; done
        fi
        # The board's factory MAC, out of prodnv, into the driver -- see wifi-mac. It has to be applied
        # before netifd starts hostapd, and before the pin below reads the address back.
        /opt/changxing60/bin/wifi-mac >>/tmp/changxing60-wifi-start.log 2>&1
        # OpenWrt derives the AP's BSSID from the phy MAC, which is all-zero on this chip, leaving
        # hostapd with bssid=00:00:00:00:00:00 and no beacon at all; pin the vif to the interface's
        # actual address so hostapd and the firmware agree on the BSSID. (That address is the factory
        # one now that wifi-mac has run, so this also keeps the BSSID stable across reboots; it is still
        # what stops the all-zero case, which is why it stays.) Before netifd starts hostapd, i.e. here.
        wifimac=$(cat /sys/class/net/wlan0/address 2>/dev/null)
        if [ -n "$wifimac" ] && [ "$wifimac" != "00:00:00:00:00:00" ]; then
                uci -q get wireless.ap0 >/dev/null && {
                        uci -q set wireless.ap0.macaddr="$wifimac"
                        uci -q commit wireless
                }
        fi
        # And re-pin the beacon-rewriting options off on every boot, whatever wrote the config: hostapd's
        # defaults make it rewrite the beacon every 6 s (BSS load) and every 50 s (channel utilization),
        # and on this chip each rewrite asserts the CP2 firmware until the radio is dead (measured
        # 2026-10-13). openwrt-wifi-config above now writes both for virgin installs; this catches every
        # install that predates that line, plus any later LuCI write that drops them.
        if uci -q get wireless.ap0 >/dev/null 2>&1; then
                for o in bss_load_update_period chan_util_avg_period; do
                        [ "$(uci -q get wireless.ap0.$o)" = "0" ] || {
                                uci -q set wireless.ap0.$o='0'
                                uci -q commit wireless
                        }
                done
        fi
        # Deliberately no `wifi down; wifi up` here. changxing60-hw is S12 and netifd is S20, so there is no
        # netifd to serve the ubus object yet and every `wifi` call answers "Command failed: Not found"
        # (four such lines, one per call, used to land in /tmp/changxing60-wifi-start.log) while the AP came up
        # fine a moment later under netifd, with the country and the pinned MAC already in place. An AP
        # that still ends up without a beacon is repaired by changxing60-post's wififix, which runs after netifd.
        #
        # Re-assert the LAN split here, after ap0 exists. net-split's only uci-defaults caller runs at S10,
        # two levels before this script creates the wireless config it reads (conflicts.md B1 -- on a virgin
        # flash that ordering alone lost the whole split). Without --apply net-split only writes uci, and
        # netifd (S20) has not read anything yet, so a fresh install lands on the same config today's
        # installs run; idempotently re-asserting it every boot makes the split the pinned state the rest of
        # this file already treats everything else as.
        sh /opt/changxing60/bin/net-split >>/tmp/changxing60-net-split.log 2>&1 || \
                logger -t changxing60-hw "net-split failed (rc=$?) -- the hotspot may have no firewall zone and no client will get an address"
}

start() {
        boot
}
#!/bin/sh /etc/rc.common
# Adopt the two runtime paths an old boot image creates, for a device whose boot_b has not been
# rebuilt since the rename.
#
# boot_b used to carry a boot image whose initramfs created /run/mu300 and attached the disk area at
# /mnt/mu300-disk, while every script in this rootfs uses the current names. This moves them -- once
# per boot, before anything else runs (START=00, ahead of changxing60-hw at S12).
#
# As of 2026-10-16 boot_b *has* been rebuilt from boot/init, which creates /run/changxing60 and
# mounts /mnt/changxing60-disk itself, so on this device both branches below are now no-ops and this
# script can be deleted. It is kept because it is the only thing that makes the rootfs rename safe to
# do *before* the boot image is rebuilt -- which is the order tools/rename-device.sh runs in, and the
# order a re-run on a fresh device would run in. See docs/changxing60/rename.md.
#
# It is not cosmetic. changxing60-next-boot --rearm reads /run/changxing60/misc-bc-slot-b-trial.bin;
# with the file absent, `set -eu` aborts the write and the slot-b re-arm that keeps Linux as the
# default boot fails silently -- the device then rolls back to Android after two boots. The disk-area
# move is what changxing60-os looks at to find the other system.
START=00

start() {
        # /run is a tmpfs the initramfs fills before switch_root, so rcS is the first chance to rename it.
        if [ -d /run/mu300 ] && [ ! -e /run/changxing60 ]; then
                mv /run/mu300 /run/changxing60
                logger -t changxing60-bootpaths "adopted /run/mu300 -> /run/changxing60 (boot image predates the rename)"
        fi

        # The disk area is a second mount of the root filesystem. A mount point is named by the path it is
        # attached at, so this is a move rather than a rename, and the directory it lands on must exist.
        # Tested through /proc/mounts and not with -d: the mkdir below leaves an empty directory behind, so
        # on the second boot a -d test would find it, call the move done, and never move anything.
        if grep -q " /mnt/mu300-disk " /proc/mounts && ! grep -q " /mnt/changxing60-disk " /proc/mounts; then
                mkdir -p /mnt/changxing60-disk
                if mount --move /mnt/mu300-disk /mnt/changxing60-disk 2>/dev/null; then
                        logger -t changxing60-bootpaths "adopted /mnt/mu300-disk -> /mnt/changxing60-disk"
                else
                        logger -t changxing60-bootpaths "could not move the disk area to /mnt/changxing60-disk; changxing60-os will not find it"
                fi
        fi
}
#!/bin/sh /etc/rc.common
# After boot: keep mobile data up across modem resets, and mark the boot successful (re-arm slot b if Linux is default)
START=99
USE_PROCD=1
export PATH="$PATH:/opt/changxing60/busybox-bin"

start_service() {
        # Why did this boot happen? Write it down. /proc/cmdline's bootcause is generated per boot by the
        # bootloader, and pstore keeps only ONE crash record, so without this line a reboot's cause is gone
        # the moment the next one lands. That is not hypothetical: on 2026-10-16 an oops in the audio path
        # (see docs/changxing60/audio-oops.md) took the device down at 18:28, and an earlier reboot at 19:49
        # in the same evening could not be explained at all -- its pstore record had already been overwritten.
        # One appended line per boot makes both answerable. Append-only, capped, and it must never be the
        # thing that fails a boot, hence the redirect and the trailing true.
        {
                hist=/etc/changxing60/boot-history
                mkdir -p /etc/changxing60 2>/dev/null
                printf '%s uptime=%s kernel=%s %s\n' \
                        "$(date '+%Y-%m-%d %H:%M:%S %z')" \
                        "$(cut -d. -f1 /proc/uptime 2>/dev/null)" \
                        "$(uname -r 2>/dev/null)" \
                        "$(grep -o 'bootcause="[^"]*"' /proc/cmdline 2>/dev/null) $(grep -o 'androidboot\.mode=[^ ]*' /proc/cmdline 2>/dev/null)" \
                        >> "$hist" 2>/dev/null
                tail -n 200 "$hist" > "$hist.new" 2>/dev/null && mv "$hist.new" "$hist" 2>/dev/null
        } 2>/dev/null
        # Last word on the console level: changxing60-hw set it before the WLAN modules loaded, but the
        # vendor modules keep loading after that (S12 is early), so re-assert it here. See quiet-console
        # for why a chatty console wedges the SC2355 firmware.
        /opt/changxing60/bin/quiet-console
        procd_open_instance cellwatch
        procd_set_param command /opt/changxing60/bin/mobile-data watch
        procd_set_param env CHANGXING60_NETIFD=1 PATH="$PATH"
        procd_set_param respawn 3600 30 0
        procd_close_instance
        procd_open_instance thermalguard
        procd_set_param command /opt/changxing60/bin/thermal-guard
        procd_set_param env PATH="$PATH"
        procd_close_instance
        # The bq2560x probe programs the end-of-charge threshold from the pack's *max charge
        # current* (charge stops at ~94%) and sizes the input current limit from USB detection
        # (0.19-0.8 A into the cell depending on the boot), so the cell never charged at its
        # ~1 A. charge-fix re-pins ITERM 180 mA / input limit 1.2 A / ICHG 960 mA / VREG
        # 4368 mV (the vendor driver's own value for this pack; the live chip was once found
        # at 4432 mV from a one-shot write of unknown origin) on the chip over i2c, and
        # applies the optional /opt/changxing60/charge-cap top-off cap (charge off at the cap,
        # back on 40 permille below; the board keeps running off VBUS either way). A
        # crontab entry re-runs it every minute against USB replugs. Waiting for
        # the i2c node is the script's job.
        procd_open_instance chargefix
        procd_set_param command /opt/changxing60/bin/charge-fix
        procd_set_param env PATH="$PATH"
        procd_close_instance
        # The SC2355 sometimes refuses the first IFF_UP (EIO), and an AP that came up while the world
        # regulatory domain was still active stays AP-DISABLED: both need a real down/up. This has to run
        # after netifd (S20) -- changxing60-hw is S12 and its wifi calls there only answer "Command failed: Not
        # found" -- so it lives here, 35 s in, with wififix deciding whether anything is wrong.
        procd_open_instance wififix
        procd_set_param command /bin/sh -c "sleep 35; exec /opt/changxing60/bin/wifi-fix"
        procd_set_param env PATH="$PATH"
        procd_close_instance
        # wifi-fix repairs the AP once, at startup. This keeps repairing it: an AP that goes off air
        # later (a firmware assert that powered the radio off under changxing60-hw's reset policy, a failed
        # CMD_START_AP) looks perfectly healthy from the host, so something has to test it the way a
        # client would and put it back. Restarts wifi rather than rebooting -- see hotspot-watch.
        procd_open_instance hotspotwatch
        procd_set_param command /opt/changxing60/bin/hotspot-watch
        procd_set_param env PATH="$PATH"
        procd_set_param respawn 3600 30 0
        procd_close_instance
        procd_open_instance bootok
        procd_set_param command /bin/sh -c "sleep 30; exec /opt/changxing60/bin/changxing60-next-boot --rearm"
        procd_set_param env PATH="$PATH"
        procd_close_instance
}
